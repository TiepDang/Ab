function exportCSV() { const dates = Object.keys(attendanceData); if (dates.length === 0) return alert('Chưa có dữ liệu!'); let csvContent = "\uFEFFNgày,Giờ Vào,Giờ Về,Số Công\n"; dates.forEach(date => { const currentVal = attendanceData[date].val !== undefined ? attendanceData[date].val : 1; // Sử dụng chính xác dấu nháy ngược ` ở đầu và cuối dòng dưới đây: csvContent += `${date},${attendanceData[date].in},${attendanceData[date].out},${currentVal}\n`; }); const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' }); const url = URL.createObjectURL(blob); const link = document.createElement("a"); link.href = url; link.download = "bang_tinh_cong.csv"; link.click(); }
